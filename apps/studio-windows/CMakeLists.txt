cmake_minimum_required(VERSION 3.20)
project(StudioLinkStudioWindows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "StudioLink Studio Windows must be built on Windows 11 with MSVC.")
endif()
add_executable(StudioLinkStudio WIN32 src/main.cpp)
target_compile_definitions(StudioLinkStudio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(StudioLinkStudio PRIVATE ole32 avrt mmdevapi)
